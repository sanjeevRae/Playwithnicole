function PageTitle({ kicker, title, sub }: { kicker: string; title: string; sub: string }) {
  return <div className="pageHead"><span className="kicker">{kicker}</span><h1>{title}</h1><p>{sub}</p></div>

function Modal({ title, children, close }: { title: string; children: React.ReactNode; close: () => void }) {
  return <div className="dialogShade" role="presentation" onMouseDown={e => e.target === e.currentTarget && close()}><section className="dialog" role="dialog" aria-modal="true"><div className="dialogHead"><h2>{title}</h2><button onClick={close} aria-label="Close"><Icon name="close"/></button></div>{children}</section></div>
}

function AuthModal({ mode, setMode, enter }: { mode: Exclude<Auth, null>; setMode: (a: Auth) => void; enter: () => void }) {
  const { signUp, signIn, signInWithGoogle, loading } = useAuth()
  const [formError, setFormError] = useState<string | null>(null)
  const [email, setEmail] = useState('')
  const [password, setPassword] = useState('')
  const [fullName, setFullName] = useState('')
  const [username, setUsername] = useState('')
  const copy = {
    login: ['Welcome back', 'Sign in to continue your adventure.'],
    signup: ['Create your account', 'Join the community and start exploring.'],
    forgot: ['Reset your password', "We'll send secure reset instructions to your inbox."],
    verify: ['Check your inbox', 'We sent a verification link to player@example.com.'],
  }[mode]

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    setFormError(null)
    try {
      if (mode === 'forgot') {
        await sendPasswordResetEmail(auth, email)
        setMode('verify')
        return
      }
      if (mode === 'signup') { await signUp(email, password, fullName) } else { await signIn(email, password) }
      enter()
    } catch (err: any) { setFormError(err?.message || 'Something went wrong. Please try again.') }
  }

  const handleGoogle = async () => {
    setFormError(null)
    try { await signInWithGoogle(); enter() } catch (err: any) { setFormError(err?.message || 'Google sign-in failed. Please try again.') }
  }

  return (
    <div className="modal">
      <div className="modalArt" style={{ backgroundImage: `url(${hero})` }}>
        <Logo/>
        <div>
          <span className="kicker">PLAY â€¢ CONNECT â€¢ DISCOVER</span>
          <h2>Your next favorite game is waiting.</h2>
          <p>One profile. A whole world of play.</p>
        </div>
      </div>
      <div className="formSide">
        <button className="modalClose" onClick={() => setMode(null)}><Icon name="close"/></button>
        <div className="formWrap">
          <span className="formLogo"><Icon name={mode === 'verify' ? 'shield' : 'game'} size={26}/></span>
          <h2>{copy[0]}</h2>
          <p>{copy[1]}</p>
          {formError && <div className="toast error" style={{ marginBottom: 16 }}>{formError}</div>}
          {mode === 'verify' ? (
            <>
              <div className="verifyArt">âœ¦</div>
              <Button onClick={enter}>Continue to dashboard</Button>
              <button className="formLink">Resend verification email</button>
            </>
          ) : (
            <form onSubmit={handleSubmit}>
              {mode === 'signup' && (
                <div className="fieldRow">
                  <label>Full name<input placeholder="Nicole Player" value={fullName} onChange={e => setFullName(e.target.value)}/></label>
                  <label>Username<input placeholder="nicoleplays" value={username} onChange={e => setUsername(e.target.value)}/></label>
                </div>
              )}
              <label>{mode === 'login' ? 'Email or username' : 'Email address'}<input type="text" placeholder={mode === 'login' ? 'you@example.com or username' : 'you@example.com'} value={email} onChange={e => setEmail(e.target.value)}/></label>
              {mode !== 'forgot' && (
                <label>Password
                  <div className="inputWrap"><input type="password" placeholder="At least 8 characters" value={password} onChange={e => setPassword(e.target.value)}/></div>
                </label>
              )}
              {mode === 'signup' && (
                <>
                  <div className="strength"><span/><span/><span/><span/></div>
                  <label className="check"><input type="checkbox"/> I agree to the Terms and Privacy Policy</label>
                </>
              )}
              {mode === 'login' && (
                <div className="formMeta">
                  <label className="check"><input type="checkbox"/> Remember me</label>
                  <button type="button" onClick={() => setMode('forgot')}>Forgot password?</button>
                </div>
              )}
              <Button disabled={loading}>{mode === 'login' ? 'Log in' : mode === 'signup' ? 'Create account' : 'Send reset link'} <Icon name="arrow" size={16}/></Button>
              {mode === 'login' && <button className="google" type="button" onClick={handleGoogle} disabled={loading}>G&nbsp;&nbsp; Continue with Google <small>visual only</small></button>}
            </form>
          )}
          <p className="switch">{mode === 'login' ? 'New to playwithnicole? ' : mode === 'signup' ? 'Already have an account? ' : ''}<button onClick={() => setMode(mode === 'login' ? 'signup' : 'login')}>{mode === 'login' ? 'Create an account' : 'Back to login'}</button></p>
        </div>
      </div>
    </div>
  )
}

function GameForm({ game, close, save }: { game?: DemoGame; close: () => void; save: (g: DemoGame) => void }) {
  const [preview, setPreview] = useState(game?.image || photos[0])
  return (
    <form className="gameForm" onSubmit={e => { e.preventDefault(); const d = new FormData(e.currentTarget); save({ id: game?.id || Date.now(), title: String(d.get('title')), category: String(d.get('category')), status: String(d.get('status')), visible: d.get('visible') === 'on', image: preview }) }}>
      <div className="uploadPreview"><img src={preview} alt="Cover"/><label>Upload cover<input type="file" accept="image/*" onChange={e => { const f = e.target.files?.[0]; if (f) setPreview(URL.createObjectURL(f)) }}/></label></div>
      <label>Game title<input name="title" required defaultValue={game?.title}/></label>
      <div className="dialogActions"><Button secondary onClick={close}>Cancel</Button><Button>{game ? 'Save changes' : 'Create game'}</Button></div>
    </form>
  )
}
